#include <Servo.h>

Servo servo1;
Servo servo2;

const int servo1Pin = 9;
const int servo2Pin = 10;

// D9 固定位置
const int servo1Angle = 90;

// D10 用脉冲控制
int pulse = 1500;
int direction = 1;

// 先不要把范围开太大
const int minPulse = 1500;
const int maxPulse = 2400;

// 控制速度
unsigned long previousMillis = 0;
const unsigned long interval = 25;

// 每次移动多少微秒
const int stepSize = 5;

void setup() {
  servo1.attach(servo1Pin);
  servo2.attach(servo2Pin);

  servo1.write(servo1Angle);
  servo2.writeMicroseconds(pulse);
}

void loop() {
  unsigned long currentMillis = millis();

  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis;

    pulse += direction * stepSize;

    if (pulse >= maxPulse) {
      pulse = maxPulse;
      direction = -1;
    }

    if (pulse <= minPulse) {
      pulse = minPulse;
      direction = 1;
    }

    servo2.writeMicroseconds(pulse);
  }
}
